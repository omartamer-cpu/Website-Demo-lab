-- Q1.3 (b) department 3 still has an employee -> error
SQL> DELETE FROM department WHERE Dnumber = 3;
ERROR 1451: Cannot delete or update a parent row: a foreign key constraint fails (`company`.`employee`, CONSTRAINT `employee_ibfk_2` FOREIGN KEY (`Dno`) REFERENCES `department` (`Dnumber`))
-- correct order (inside a transaction, rolled back so the data stays for Part B):
-- 1) children first: move (or delete) the employees of the department
-- 2) then delete the department
SQL> START TRANSACTION;
OK, 0 row(s) affected
SQL> UPDATE employee SET Dno = 2 WHERE Dno = 3;
OK, 1 row(s) affected
SQL> DELETE FROM department WHERE Dnumber = 3;
OK, 1 row(s) affected
SQL> SELECT Dnumber, Dname FROM department;
+---------+-----------+
| Dnumber | Dname     |
+---------+-----------+
| 1       | HR        |
| 2       | IT        |
| 4       | Marketing |
+---------+-----------+
3 row(s)
SQL> ROLLBACK;
OK, 0 row(s) affected
SQL> SELECT Dnumber, Dname FROM department;
+---------+-----------+
| Dnumber | Dname     |
+---------+-----------+
| 1       | HR        |
| 2       | IT        |
| 4       | Marketing |
| 3       | Sales     |
+---------+-----------+
4 row(s)

| 2   | 3         |

-- WHERE vs HAVING: WHERE filters single rows before grouping (no aggregates);
-- HAVING filters whole groups after grouping and can use aggregates such as COUNT(*).
